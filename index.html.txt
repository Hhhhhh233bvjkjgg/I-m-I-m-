  <script>
    const defaultProducts = [

        id: 1,

      },
      {

        category: "بخور",
        price: 89,
        image: "https://images.unsplash.com/photo-1603006905003-be475563bc59?auto=format&fit=crop&w=800&q=80",
        description: "بخور فاخر برائحة ثابتة ومميزة."
      },
      {
        id: 3,
        name: "زيت العود",
        category: "زيوت",
        price: 159,
        image: "https://images.unsplash.com/photo-1615634260167-c8cdede054de?auto=format&fit=crop&w=800&q=80",
        description: "زيت عود مركز للاستخدام اليومي والمناسبات."
      }
    ];

    let products = JSON.parse(localStorage.getItem("biyof-products")) || defaultProducts;
    let cart = JSON.parse(localStorage.getItem("biyof-cart")) || [];

    function save() {
      localStorage.setItem("biyof-products", JSON.stringify(products));
      localStorage.setItem("biyof-cart", JSON.stringify(cart));
    }

    function showPage(page) {
      document.querySelectorAll(".page").forEach(item => {
        item.classList.remove("active");
      });

      document.getElementById(page).classList.add("active");
      document.getElementById("links").classList.remove("show");

      renderProducts();
      renderAdmin();

      window.scrollTo({
        top: 0,
        behavior: "smooth"
      });
    }

    function toggleMenu() {
      document.getElementById("links").classList.toggle("show");
    }

    function productHTML(product) {
      return `
        <article class="product">
          <div class="product-img">
            <img src="${product.image}" alt="${product.name}">
          </div>

          <div class="product-info">
            <small>${product.category}</small>
            <h3>${product.name}</h3>
            <p>${product.description || ""}</p>

            <div class="product-bottom">
              <strong>${product.price} ر.س</strong>
              <button class="add" onclick="addToCart(${product.id})">
                إضافة للسلة
              </button>
            </div>
          </div>
        </article>
      `;
    }

    function renderProducts() {
      const home = document.getElementById("homeProducts");
      const all = document.getElementById("allProducts");
      const search = document.getElementById("search")?.value.toLowerCase() || "";
      const selected = document.getElementById("category")?.value || "all";

      const categories = [...new Set(products.map(product => product.category))];
      const categoryElement = document.getElementById("category");

      if (categoryElement) {
        const current = categoryElement.value;
        categoryElement.innerHTML =
          `<option value="all">كل التصنيفات</option>` +
          categories.map(category => `<option value="${category}">${category}</option>`).join("");
        categoryElement.value = categories.includes(current) ? current : "all";
      }

      const filtered = products.filter(product => {
        const matchesSearch = product.name.toLowerCase().includes(search);
        const matchesCategory = selected === "all" || product.category === selected;
        return matchesSearch && matchesCategory;
      });

      if (home) {
        home.innerHTML = products.slice(0, 3).map(productHTML).join("");
      }

      if (all) {
        all.innerHTML = filtered.length
          ? filtered.map(productHTML).join("")
          : "<p>لا توجد منتجات.</p>";
      }
    }

    function addToCart(id) {
      const product = products.find(item => item.id === id);
      const existing = cart.find(item => item.id === id);

      if (existing) {
        existing.quantity++;
      } else {
        cart.push({ ...product, quantity: 1 });
      }

      save();
      renderCart();
      openCart();
    }

    function renderCart() {
      const count = cart.reduce((sum, item) => sum + item.quantity, 0);
      const total = cart.reduce((sum, item) => sum + item.price * item.quantity, 0);

      document.getElementById("cartCount").textContent = count;
      document.getElementById("cartTotal").textContent = total + " ر.س";

      const box = document.getElementById("cartItems");

      if (!cart.length) {
        box.innerHTML = "<p style='padding:30px 0;color:#806e60'>السلة فارغة</p>";
        return;
      }

      box.innerHTML = cart.map(item => `
        <div class="cart-item">
          <img src="${item.image}" alt="${item.name}">
          <div>
            <h3>${item.name}</h3>
            <p>${item.quantity} × ${item.price} ر.س</p>
          </div>
          <button class="remove" onclick="removeFromCart(${item.id})">حذف</button>
        </div>
      `).join("");
    }

    function removeFromCart(id) {
      cart = cart.filter(item => item.id !== id);
      save();
      renderCart();
    }

    function openCart() {
      document.getElementById("overlay").classList.add("show");
      renderCart();
    }

    function closeCart(event) {
      if (!event || event.target.id === "overlay") {
        document.getElementById("overlay").classList.remove("show");
      }
    }

    function addProduct(event) {
      event.preventDefault();

      const form = new FormData(event.target);

      products.unshift({
        id: Date.now(),
        name: form.get("name"),
        category: form.get("category"),
        price: Number(form.get("price")),
        image: form.get("image") || "https://images.unsplash.com/photo-1547887538-e3a2f32cb1cc?auto=format&fit=crop&w=800&q=80",
        description: form.get("description")
      });

      save();
      event.target.reset();
      renderProducts();
      renderAdmin();
      alert("تمت إضافة المنتج بنجاح");
    }

    function renderAdmin() {
      const list = document.getElementById("adminList");
      if (!list) return;

      list.innerHTML = products.map(product => `
        <div class="admin-item">
          <img src="${product.image}" alt="${product.name}">
          <div>
            <h3>${product.name}</h3>
            <p>${product.category}</p>
            <strong>${product.price} ر.س</strong>
          </div>
          <button class="delete" onclick="deleteProduct(${product.id})">
            حذف
          </button>
        </div>
      `).join("");
    }

    function deleteProduct(id) {
      if (!confirm("هل تريد حذف المنتج؟")) return;

      products = products.filter(product => product.id !== id);
      save();
      renderProducts();
      renderAdmin();
    }

    function checkout() {
      if (!cart.length) {
        alert("السلة فارغة");
        return;
      }

      alert("تم استلام طلبك. سيتم التواصل معك قريبًا.");
    }

    renderProducts();
    renderAdmin();
    renderCart();
  </script>
